pipeline  {
  agent any
  stages {
    stage ("clean") {
    steps {
      echo "cleaning"
      sh 'mvn clean'
    }
  }
  stage ("packing") {
        steps {
          echo "packing"
          sh 'mvn package'
        }
  }
    stage ("build") {
      steps {
        echo "building"
        sh 'mvn install'
            }
}
    stage ("Deploy") {
      steps {
        echo "deploying"
        deploy adapters: [tomcat9(alternativeDeploymentContext: '', credentialsId: 'tomcatserver', path: '', url: 'http://localhost:8080/')], contextPath: null, war: '**/*.war'
       }    
    }
  }
}
